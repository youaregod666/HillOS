# HillOS
it's a opencomputers operating system based on MineOS
# Install UEFI or the OS
UEFI/BIOS

	wget -f https://raw.githubusercontent.com/youaregod666/HillOS/Dev/EFI/Minified.lua /tmp/efi.lua && flash -q /tmp/efi.lua && reboot

Instaler

	wget -f https://raw.githubusercontent.com/youaregod666/HillOS/Dev/Installer/BIOS.lua /tmp/bios.lua && flash -q /tmp/bios.lua && reboot

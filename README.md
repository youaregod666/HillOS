# HillOS
it's a opencomputers operating system based on MineOS
# Install UEFI or the OS
UEFI/BIOS

	wget -f https://raw.githubusercontent.com/youaregod666/HillOS/Dev_LongCopenhagen/EFI/Minified.lua /tmp/efi.lua && flash -q /tmp/efi.lua && reboot

Installer

	wget -f https://raw.githubusercontent.com/youaregod666/HillOS/Dev_LongCopenhagen/Installer/BIOS.lua /tmp/bios.lua && flash -q /tmp/bios.lua && reboot
